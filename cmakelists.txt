cmake_minimum_required(VERSION 3.16)
project(FileCompressionTool VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find Qt6
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)

# Enable automoc for Qt signals/slots
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

# Source files
set(SOURCES
    main.cpp
    src/Huffman.cpp
    src/RunLength.cpp
    src/CompressionManager.cpp
    src/Utility.cpp
    src/ReportGenerator.cpp
    src/MainWindow.cpp
)

# Header files  
set(HEADERS
    include/Huffman.h
    include/RunLength.h
    include/CompressionManager.h
    include/Utility.h
    include/ReportGenerator.h
    include/MainWindow.h
)

# UI files - FIXED PATH
set(FORMS
    gui/mainwindow.ui
)

# Create executable
add_executable(FileCompressionTool 
    ${SOURCES}
    ${HEADERS}
    ${FORMS}
)

# Include directories
target_include_directories(FileCompressionTool PRIVATE include)

# Link Qt libraries
target_link_libraries(FileCompressionTool PRIVATE Qt6::Core Qt6::Widgets)

# Explicitly tell CMake where to find UI files
set(CMAKE_AUTOUIC_SEARCH_PATHS "gui")
